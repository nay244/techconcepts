import os, re, sys
old, new = sys.argv[1], sys.argv[2]
CARD = ['index.html', 'tracks/index.html', 'how-the-internet-works/index.html', 'networking/index.html', 'about/index.html', 'glossary/index.html', '404.html']
UID = re.compile(r'\buid="[^"]*"')
CS = re.compile(r'/_astro/card-stack\.[\w-]+\.css')
RULE = '@media (width<=959px){html:has(#menu-panel:not([hidden])){scroll-padding-top:341px}}'
def walk(d): return {os.path.relpath(os.path.join(r, f), d) for r, _, fs in os.walk(d) for f in fs}
def read(d, f): return open(os.path.join(d, f), 'rb').read()
def cascade(d, html):  # every stylesheet of the page, inline or linked, in document order, as one text
    out = []
    for m in re.finditer(r'<link rel="stylesheet" href="([^"]+)"[^>]*>|<style>(.*?)</style>', html, re.S):
        out.append(read(d, m.group(1).lstrip('/')).decode() if m.group(1) else m.group(2))
    return '\n'.join(t.strip() for t in out)
bad = []
fo, fn = walk(old), walk(new)
added, gone = sorted(fn - fo), sorted(fo - fn)
cs_new = [f for f in added if re.fullmatch(r'_astro/card-stack\.[\w-]+\.css', f)]
tr_new = [f for f in added if re.fullmatch(r'_astro/tracks\.[\w-]+\.css', f)]
tr_old = [f for f in gone if re.fullmatch(r'_astro/tracks\.[\w-]+\.css', f)]
for f in added:
    if f not in cs_new + tr_new: bad.append('new file: ' + f)
for f in gone:
    if f not in tr_old: bad.append('file gone: ' + f)
if len(cs_new) > 1: bad.append('more than one card-stack stylesheet')
if tr_new or tr_old:
    if len(tr_new) != 1 or len(tr_old) != 1: bad.append('Tracks stylesheet: expected one swapped')
    else:
        a, b = read(old, tr_old[0]).decode(), read(new, tr_new[0]).decode()
        if b.replace(RULE, '', 1) != a or RULE not in b: bad.append('Tracks stylesheet differs by more than the 341px rule')
for f in sorted(fo & fn):

        if re.findall(r'<main\b.*?</main>', sx, re.S) != re.findall(r'<main\b.*?</main>', sy, re.S): bad.append('main differs: ' + f)
        if sx.split('</head>', 1)[1] != sy.split('</head>', 1)[1]: bad.append('body differs (header, footer or main): ' + f)
        SHEET = r'<link rel="stylesheet" href="[^"]+"[^>]*>|<style>.*?</style>'
        if re.sub(SHEET, '', sx.split('</head>')[0], flags=re.S) != re.sub(SHEET, '', sy.split('</head>')[0], flags=re.S): bad.append('head differs outside its stylesheets: ' + f)
        if sx.split('<main', 1)[0].count('<script') != sy.split('<main', 1)[0].count('<script') or sx.count('<script') != sy.count('<script'): bad.append('script count differs: ' + f)
